Hotspot m0
    Attributes
        ID                                  : 31
        Scope                               : local
        Dragging                            : 0
        Disable release event after dragging: 0
        Send Component ID                   : disabled
    
    Events
        Touch Press Event
            // Update all components on page
            // First, set the right background/crop images
            if(statusl==2||statusl==3)
            {
                coolingl.val=1
            }else
            {
                coolingl.val=0
            }
            if(statusb==2||statusb==3)
            {
                coolingb.val=1
            }else
            {
                coolingb.val=0
            }
            if(coolingl.val==1&&coolingb.val==0)
            {
                // Living room cooling is ON, Bedroom cooling is OFF
                va1.val=6
            }else if(coolingl.val==0&&coolingb.val==1)
            {
                // Bedroom cooling is ON, living rooom cooling is OFF
                va1.val=4
            }else if(coolingl.val==0&&coolingb.val==0)
            {
                // No cooling
                va1.val=0
            }else
            {
                // both bed and living room cooling is on
                va1.val=2
            }
            // Update page background if necessary
            if(sys0!=va1.val)
            {
                Home.pic=va1.val+1
                sys0=va1.val
            }
            // Update thermostat mode buttons
            if(model==1)
            {
                click q0,1
            }else if(model==2)
            {
                click q1,1
            }else
            {
                click q2,1
            }
            if(modeb==1)
            {
                click q4,1
            }else if(modeb==2)
            {
                click q5,1
            }else
            {
                click q6,1
            }
            // Update target temperature fields
            covx targettempl,ttl.txt,0,0
            ttl.txt+=" ºF"
            ttl.picc=va1.val
            covx targettempb,ttb.txt,0,0
            ttb.txt+=" ºF"
            ttb.picc=va1.val
            // Update AC status
            spstr ",Idle,Cooling (low),Cooling (high),Fanning",sb.txt,",",statusb
            spstr ",Idle,Cooling (low),Cooling (high),Fanning",sl.txt,",",statusl
            sb.picc=va1.val
            sl.picc=va1.val
            //Update temperature fields
            tl.picc=va1.val
            hl.picc=va1.val
            tb.picc=va1.val
            hb.picc=va1.val

    Events
        Touch Press Event
            dim=brightness.val
            dimtimer.en=1
            sleeptimer.en=1
            swipestore.en=1
        
        Touch Release Event
            swipestore.en=0
            // Touch has ended, x
            if(tch0==0)
            {
                swipec=swipex-tch2
                // From left to right
                if(swipec>swipedx)
                {
                    //go to page on the left - disabled
                    // page Home
                    // hometimer.en=0 (only needed when going to Home)
                }
                // From right to left
                swipec2=0-swipedx
                if(swipec<swipec2)
                {
                    // go to page on the right - disabled
                    //  page Home
                    // hometimer.en=0 (only needed when going to Home)
                }
            }
            // Touch has ended, y
            if(tch1==0)
            {
                swipec=swipey-tch3
                // Pull down - From top to bottom
                if(swipec>100)
                {
                    // Pull down page from the top
                    page Internet
                }
                // Pull up - From bottom to top
                swipec2=0-swipedy
                if(swipec<swipec2)
                {
                    // Pull up page from the bottom
                    page Internet
                }
            }
